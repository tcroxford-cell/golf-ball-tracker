Auto-putt: {putts} putts. Press N for next hole." # ========================== # COURSE SETUP # ========================== def create_sample_course(): holes = [] for i in range(9): par = 3 if i % 3 == 0 else 4 tee_x = 100 tee_y = 500 pin_x = 900 pin_y = 300 fairway_rect = pygame.Rect(50, 350, 1000, 200) green_rect = pygame.Rect(850, 250, 150, 100) bunker_rects = [ pygame.Rect(500, 400, 80, 60), pygame.Rect(700, 380, 60, 40), ] hole = Hole( index=i + 1, par=par, tee_pos=(tee_x, tee_y), pin_pos=(pin_x, pin_y), fairway_rect=fairway_rect, green_rect=green_rect, bunker_rects=bunker_rects, ) holes.append(hole) return holes # ========================== # COLLISION / ZONES # ========================== def point_in_rect(x, y, rect: pygame.Rect): return rect.collidepoint(x, y) def check_landing_zone(game_state: GameState): hole = game_state.current_hole ball = game_state.ball if point_in_rect(ball.x, ball.y, hole.green_rect): game_state.message = "Ball on green!
